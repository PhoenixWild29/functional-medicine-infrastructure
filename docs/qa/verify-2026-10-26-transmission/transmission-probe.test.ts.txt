/**
 * @jest-environment node
 *
 * THROWAWAY probe for D-2026-10-26-04 (never committed). Runs the real
 * signature binding and send guard of order-transmission-integrity
 * e2d293e1 against scripted rows.
 */

import { createHash, createHmac } from 'node:crypto'
import { scriptedDb } from '@/__tests__/helpers/scripted-db'
import {
  LEGACY_SIGNING_CUTOFF,
  boundSignatureValue,
  canonicalSignedContent,
  patientSnapshotFor,
  prescriberSnapshotFor,
  rerouteBinding,
  verifySignatureBinding,
  type SignedOrderRow,
} from '@/lib/orders/signature-binding'
import { loadTransmittableOrder } from '@/lib/adapters/transmission-guard'

const ORDER = '11111111-1111-4111-8111-111111111111'
const PH_A = 'aaaaaaaa-aaaa-4aaa-8aaa-aaaaaaaaaaaa'
const PH_B = 'bbbbbbbb-bbbb-4bbb-8bbb-bbbbbbbbbbbb'
const DIGEST = 'd'.repeat(64)
const LOCKED = '2026-10-26T12:00:00.000Z'
const PATIENT = { first_name: 'Jane', last_name: 'Doe', date_of_birth: '1980-01-01', allergies: ['penicillin'], nkda: false }

function baseOrder(over: Partial<SignedOrderRow> = {}): SignedOrderRow & Record<string, unknown> {
  return {
    order_id: ORDER, status: 'SUBMISSION_PENDING', locked_at: LOCKED, pharmacy_id: PH_A, reroute_count: 0,
    patient_id: 'pt-1', provider_id: 'pr-1',
    patient_snapshot: patientSnapshotFor(PATIENT),
    prescriber_snapshot: prescriberSnapshotFor({ provider_id: 'pr-1', first_name: 'A', last_name: 'B', npi_number: '1234567890', dea_number: 'AB1234567', license_state: 'TX' }),
    provider_npi_snapshot: '1234567890',
    pharmacy_snapshot: { pharmacy_id: PH_A, name: 'A' },
    medication_snapshot: { medication_name: 'X' },
    sig_text: 'one daily', quantity: 1, refills: 0,
    ...over,
  }
}
const signed = (o: SignedOrderRow) => ({ ...o, provider_signature_hash_snapshot: boundSignatureValue(DIGEST, LOCKED, canonicalSignedContent(o)) })

function dbFor(order: unknown, history: unknown[] = []) {
  return scriptedDb(c => {
    if (c.table === 'orders') return { data: order }
    if (c.table === 'order_status_history') return { data: history }
    if (c.table === 'patients') return { data: { ...PATIENT, address_line1: null, address_line2: null, city: null, state: null, zip: null } }
    if (c.table === 'providers') return { data: { first_name: 'A', last_name: 'B', npi_number: '1', dea_number: null, license_state: 'TX' } }
    return { data: null }
  }).client
}
const send = (order: unknown, history: unknown[] = [], opts = {}) => loadTransmittableOrder(dbFor(order, history), ORDER, opts)
const code = async (p: Promise<unknown>) => p.then(() => 'SENT', (e: { code?: string; message: string }) => e.code ?? `error: ${e.message}`)

const env = { ...process.env }
afterEach(() => { process.env = { ...env } })
beforeEach(() => { process.env['RX_BINDING_SECRET'] = 'the-real-server-secret'; delete process.env['VERCEL_ENV'] })

describe('T1', () => {
  it('a correctly signed order is sent', async () => {
    expect(await code(send(signed(baseOrder())))).toBe('SENT')
  })

  it('a value computed without the server secret is refused (unkeyed SHA-256, the public development key, the v2 form)', async () => {
    const o = baseOrder({ sig_text: 'altered', refills: 11 })
    const content = canonicalSignedContent(o)
    const msg = `compoundiq/rx-signature-binding/v3\n${DIGEST}\n${LOCKED}\n${content}`
    const unkeyed = createHash('sha256').update(msg).digest('hex')
    const devKey = createHmac('sha256', 'compoundiq-development-rx-binding-key-not-for-production').update(msg).digest('hex')
    for (const v of [`v3:${DIGEST}:${unkeyed}`, `v3:${DIGEST}:${devKey}`, `v2:${DIGEST}:${unkeyed}`, `v3:${DIGEST}`]) {
      expect(await code(send({ ...o, provider_signature_hash_snapshot: v }))).toBe('signature_mismatch')
    }
  })

  it('altered content after a real signature is refused', async () => {
    const o = signed(baseOrder())
    expect(await code(send({ ...o, sig_text: 'two daily' }))).toBe('signature_mismatch')
  })

  it('64 zeros (any bare digest) is refused without a signing history row', async () => {
    expect(await code(send(baseOrder({ provider_signature_hash_snapshot: '0'.repeat(64) })))).toBe('signature_mismatch')
  })

  it('legacy is granted only by a pre-cutoff signing row; batch-sign rows must carry the same digest', async () => {
    const zero = '0'.repeat(64)
    const o = baseOrder({ provider_signature_hash_snapshot: zero })
    const row = (meta: Record<string, unknown>, at = '2026-10-20T00:00:00Z') => ({ old_status: 'DRAFT', new_status: 'AWAITING_PAYMENT', created_at: at, metadata: meta })
    expect(await code(send(o, [row({ actor: 'provider_batch_sign', signature_hash: zero })]))).toBe('SENT')
    expect(await code(send(o, [row({ actor: 'provider_batch_sign', signature_hash: 'f'.repeat(64) })]))).toBe('signature_mismatch')
    expect(await code(send(o, [row({ actor: 'provider_batch_sign', signature_hash: zero }, '2026-11-02T00:00:00Z')]))).toBe('signature_mismatch')
    expect(await code(send(o, [row({ actor: 'provider_batch_sign', signature_hash: zero, signature_binding: 'v3' })]))).toBe('signature_mismatch')
    expect(await code(send(o, [row({ actor: 'someone_else', signature_hash: zero })]))).toBe('signature_mismatch')
    // provider_sign_and_send rows are not compared with the column: any bare digest passes with one.
    expect(await code(send(o, [row({ actor: 'provider_sign_and_send' })]))).toBe('SENT')
    expect(LEGACY_SIGNING_CUTOFF).toBe('2026-11-01T00:00:00.000Z')
  })

  it('no key in production: refused with binding_key_missing; the message names the variable, not a value', async () => {
    const o = signed(baseOrder())
    delete process.env['RX_BINDING_SECRET']
    process.env['VERCEL_ENV'] = 'production'
    const err = await send(o).catch(e => e as { code: string; message: string })
    expect(err.code).toBe('binding_key_missing')
    expect(err.message).toContain('RX_BINDING_SECRET')
    expect(err.message).not.toContain('the-real-server-secret')
  })

  it('no key outside production: the public development key is used, so a value made with it is accepted', async () => {
    delete process.env['RX_BINDING_SECRET']
    process.env['VERCEL_ENV'] = 'preview'
    const o = baseOrder({ sig_text: 'altered' })
    const msg = `compoundiq/rx-signature-binding/v3\n${DIGEST}\n${LOCKED}\n${canonicalSignedContent(o)}`
    const devKey = createHmac('sha256', 'compoundiq-development-rx-binding-key-not-for-production').update(msg).digest('hex')
    expect(verifySignatureBinding({ ...o, provider_signature_hash_snapshot: `v3:${DIGEST}:${devKey}` })).toBe('match')
  })

  it('rotating the secret: an order signed under the old key is refused under the new one', async () => {
    const o = signed(baseOrder())
    process.env['RX_BINDING_SECRET'] = 'a-new-secret'
    expect(await code(send(o))).toBe('signature_mismatch')
  })
})

describe('T2', () => {
  it('pharmacy_id differing from the signed pharmacy is refused without a reroute proof', async () => {
    const o = signed(baseOrder())
    expect(await code(send({ ...o, pharmacy_id: PH_B }))).toBe('pharmacy_mismatch')
  })

  it('a caller naming another pharmacy is refused', async () => {
    expect(await code(send(signed(baseOrder()), [], { pharmacyId: PH_B }))).toBe('pharmacy_mismatch')
  })

  it('a reroute proof keyed by the server secret is accepted; one made with another key, or for another pharmacy, is not', async () => {
    const o = { ...signed(baseOrder()), pharmacy_id: PH_B, reroute_count: 1 }
    const reroute = (binding: string, np = PH_B) => ({ old_status: 'SUBMISSION_FAILED', new_status: 'REROUTE_PENDING', created_at: '2026-10-26T13:00:00Z', metadata: { new_pharmacy_id: np, reroute_count: 1, reroute_binding: binding } })
    const good = rerouteBinding(ORDER, o.provider_signature_hash_snapshot, PH_B, 1)
    expect(await code(send(o, [reroute(good)]))).toBe('SENT')
    const forged = createHmac('sha256', 'compoundiq-development-rx-binding-key-not-for-production')
      .update(`compoundiq/rx-reroute/v1\n${ORDER}\n${o.provider_signature_hash_snapshot}\n${PH_B}\n1`).digest('hex')
    expect(await code(send(o, [reroute(forged)]))).toBe('pharmacy_mismatch')
    expect(await code(send({ ...o, pharmacy_id: 'cccccccc-cccc-4ccc-8ccc-cccccccccccc' }, [reroute(good)]))).toBe('pharmacy_mismatch')
  })

  it('patient and prescriber come from the signing snapshots, not live rows', async () => {
    const o = signed(baseOrder())
    const sent = await send(o)
    expect(sent.patient).toEqual(expect.objectContaining({ first_name: 'Jane', last_name: 'Doe', date_of_birth: '1980-01-01' }))
    expect(sent.prescriber).toEqual(expect.objectContaining({ first_name: 'A', last_name: 'B', dea_number: 'AB1234567', license_state: 'TX', npi_number: '1234567890' }))
    // The live provider row in dbFor says npi 1 and dea null: not used.
    expect(sent.prescriber.dea_number).not.toBeNull()
  })

  it('a changed snapshot (patient name after signing) is a mismatch', async () => {
    const o = signed(baseOrder())
    expect(await code(send({ ...o, patient_snapshot: { ...(o.patient_snapshot as object), first_name: 'Janet' } }))).toBe('signature_mismatch')
  })
})

describe('T6', () => {
  it('a live allergy change after signing is refused with its own code', async () => {
    const o = signed(baseOrder())
    const db = scriptedDb(c => {
      if (c.table === 'orders') return { data: o }
      if (c.table === 'patients') return { data: { ...PATIENT, allergies: ['penicillin', 'sulfa'] } }
      return { data: [] }
    }).client
    expect(await code(loadTransmittableOrder(db, ORDER))).toBe('allergies_changed')
  })
})
